2. Page d'accueil : Top Players par continent (Filtre par région, ex: 'EUW') :
SELECT i.game_name, i.tag_line, i.icone_id, i.solo_rang, i.solo_division, i.solo_lp 
FROM invocateurs i
INNER JOIN regions r ON i.id_region = r.id_region
WHERE r.nom_region = 'EUW'
ORDER BY i.solo_lp DESC
LIMIT 10;


3. Requête complexe réglementaire (4 jointures, 1 CTE, GROUP BY + HAVING) :
WITH RegionsActives AS (
    SELECT id_region, nom_region FROM regions WHERE statut = 'actif'
)
SELECT 
    u.pseudo_site,
    reg.nom_region,
    COUNT(f.id_favori) AS total_joueurs_suivis,
    AVG(i.solo_lp) AS niveau_moyen_lp
FROM utilisateurs u
INNER JOIN groupes g ON u.id_groupe = g.id_groupe
INNER JOIN favoris f ON u.id_utilisateur = f.id_utilisateur
INNER JOIN invocateurs i ON f.invocateur_puuid = i.puuid
INNER JOIN RegionsActives reg ON i.id_region = reg.id_region
WHERE g.nom_groupe = 'Premium'
GROUP BY u.pseudo_site, reg.nom_region
HAVING COUNT(f.id_favori) >= 1;

3. Page d'accueil : Framework d'agrégation (Calcul dynamique du Winrate par Champion) :
db.matchs.aggregate([
  { $unwind: "$participants" },
  { $group: {
      _id: "$participants.championName",
      totalMatchs: { $sum: 1 },
      totalVictoires: { $sum: { $cond: [ "$participants.win", 1, 0 ] } }
    }
  },
  { $project: {
      champion: "$_id",
      winrate: { $multiply: [ { $divide: [ "$totalVictoires", "$totalMatchs" ] }, 100 ] }
    }
  },
  { $sort: { totalMatchs: -1 } }
]);
